To begin, make sure 3D mode is enabled.

NetworkTableInstance.getDefault().getTable("photonvision").getSubTable("FrontCamera")
    .getEntry("hasTarget);
    .getEntry("targetPixelsX"); // Lateral distance of the tag seen by the camera
    .getEntry("targetPixelsY"); // Vertical distance of the tag seen by the camera
    .getEntry("targetPose");
        -> .getEntry("targetPose")[0]; // Lateral pose
        -> .getEntry("targetPose")[1]; // Vertical pose
        -> .getEntry("targetPose")[2]; // Distance pose
